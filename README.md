Requirements:

 - Folia 1.21.11 or newer
 - Java 21 or newer
 - PlaceholderAPI (optional)


Configuration: 

scores:
  hidden: false

display:
  title:
    frames:
      - '&e&lWONDERLANDS'
    interval: 1000

  line-1:
    frames:
      - ' '
    interval: 1000

  line-2:
    frames:
      - ' '
    interval: 1000

  line-3:
    frames:
      - '&fPlayers: &a%server_online%'
    interval: 20
